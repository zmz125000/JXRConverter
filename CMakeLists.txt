cmake_minimum_required(VERSION 3.22)
project(JXRConverter VERSION 0.2.3 LANGUAGES C CXX)
option(JXRC_DEPLOY_QT "Bundle Qt runtime dependencies when installing the GUI" OFF)
if(JXRC_DEPLOY_QT)
    # Portable desktop archives launch directly from the extracted root.
    set(CMAKE_INSTALL_BINDIR ".")
endif()
include(GNUInstallDirs)
configure_file(cmake/version.hpp.in generated/jxrc/version.hpp @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    # Use the DLL CRT consistently, including the separately built JPEG codec.
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL" CACHE STRING "MSVC runtime library")
    add_compile_definitions(NOMINMAX _CRT_SECURE_NO_WARNINGS)
endif()
option(JXRC_BUILD_GUI "Build the Qt desktop image converter" OFF)
option(JXRC_BUILD_TESTS "Build conversion integration tests" ON)
include(cmake/Dependencies.cmake)
find_package(Threads REQUIRED)
add_library(skcms STATIC "${libjxl_SOURCE_DIR}/third_party/skcms/skcms.cc" "${libjxl_SOURCE_DIR}/third_party/skcms/src/skcms_TransformBaseline.cc")
target_compile_definitions(skcms PRIVATE SKCMS_DISABLE_HSW SKCMS_DISABLE_SKX)
option(JXRC_SHARED_ENGINE "Share the codec engine between GUI and CLI" ON)
if(JXRC_SHARED_ENGINE)
    set(jxrc_library_type SHARED)
else()
    set(jxrc_library_type STATIC)
endif()
add_library(jxrc ${jxrc_library_type} src/converter.cpp src/inspect.cpp src/native/hdr_convert.cpp src/native/jxl_hdr.cpp
    src/native/avif_hdr.cpp src/native/still_jxl.cpp src/native/still_avif.cpp
    src/native/source_export.cpp src/native/raw_still.cpp src/native/regular_decode.cpp src/native/thread_budget.cpp)
add_library(jxrc::jxrc ALIAS jxrc)
set_target_properties(jxrc PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON VERSION ${PROJECT_VERSION} SOVERSION 0)
if(JXRC_SHARED_ENGINE)
    target_compile_definitions(jxrc PRIVATE JXRC_BUILDING_LIBRARY)
    foreach(target core jxrc_raw)
        target_compile_definitions(${target} PRIVATE JXRC_BUILDING_LIBRARY)
    endforeach()
else()
    target_compile_definitions(jxrc PUBLIC JXRC_STATIC)
    foreach(target core jxrc_raw)
        target_compile_definitions(${target} PRIVATE JXRC_STATIC)
    endforeach()
endif()
foreach(target core jxrc_raw)
    target_include_directories(${target} PRIVATE "${PROJECT_SOURCE_DIR}/include")
endforeach()
target_include_directories(jxrc PUBLIC include "${CMAKE_CURRENT_BINARY_DIR}/generated" PRIVATE src/native
    "${libultrahdr_SOURCE_DIR}" "${libultrahdr_SOURCE_DIR}/lib/include"
    "${jpeg_SOURCE_DIR}/src" "${jpeg_BINARY_DIR}" "${libjxl_SOURCE_DIR}/third_party/skcms" "${stb_SOURCE_DIR}" "${libwebp_SOURCE_DIR}/src")
add_dependencies(jxrc jxrc_jpeg_build)
add_dependencies(jxrc_raw jxrc_jpeg_build)
target_compile_definitions(jxrc PRIVATE EHVIEWER_HDR_CODECS)
target_link_libraries(jxrc PRIVATE jxrc_uhdr jpeg-static jxl jxl_threads jxl_cms skcms avif jxrc_raw jxrc_jxr png_static webpdemux tiff Threads::Threads)
add_executable(jxrconvert src/main.cpp)
target_link_libraries(jxrconvert PRIVATE jxrc)
if(APPLE)
    if(JXRC_BUILD_GUI)
        set(jxrc_engine_install_dir "JXRConverter.app/Contents/Frameworks")
        file(RELATIVE_PATH jxrc_mac_engine_relative "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}" "${CMAKE_INSTALL_PREFIX}/${jxrc_engine_install_dir}")
        set_target_properties(jxrconvert PROPERTIES INSTALL_RPATH "@executable_path/${jxrc_mac_engine_relative}")
    else()
        set(jxrc_engine_install_dir "${CMAKE_INSTALL_LIBDIR}")
        set_target_properties(jxrconvert PROPERTIES INSTALL_RPATH "@executable_path/../${CMAKE_INSTALL_LIBDIR};@executable_path/${CMAKE_INSTALL_LIBDIR}")
    endif()
else()
    set(jxrc_engine_install_dir "${CMAKE_INSTALL_LIBDIR}")
    if(UNIX)
        file(RELATIVE_PATH jxrc_cli_libdir "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}" "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}")
        set_target_properties(jxrconvert PROPERTIES INSTALL_RPATH "$ORIGIN/${jxrc_cli_libdir}")
    endif()
endif()
if(JXRC_SHARED_ENGINE AND NOT (APPLE AND JXRC_BUILD_GUI))
    install(TARGETS jxrc RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT Runtime LIBRARY DESTINATION ${jxrc_engine_install_dir} COMPONENT Runtime NAMELINK_SKIP)
endif()
if(WIN32)
    enable_language(RC)
    configure_file(cmake/version.rc.in generated/version.rc @ONLY)
    target_sources(jxrconvert PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/version.rc")
endif()
if(JXRC_BUILD_TESTS)
    enable_testing()
    add_executable(jxrc_tests tests/conversion.cpp)
    target_link_libraries(jxrc_tests PRIVATE jxrc jxl avif jxrc_uhdr)
    target_include_directories(jxrc_tests PRIVATE src/native "${libultrahdr_SOURCE_DIR}" "${libavif_SOURCE_DIR}/include")
    add_test(NAME conversion COMMAND jxrc_tests)
    add_test(NAME hdr_png COMMAND jxrc_tests --hdr-png)
    add_test(NAME cli_help COMMAND jxrconvert --help)
    add_test(NAME cli_version COMMAND jxrconvert --version)
    set_tests_properties(cli_version PROPERTIES PASS_REGULAR_EXPRESSION "^${PROJECT_VERSION}\n")
    add_test(NAME cli_fixture COMMAND jxrc_tests --fixture "${CMAKE_CURRENT_BINARY_DIR}/cli-input.jpg")
    set_tests_properties(cli_fixture PROPERTIES FIXTURES_SETUP cli_input)
    add_test(NAME cli_convert COMMAND jxrconvert
        "${CMAKE_CURRENT_BINARY_DIR}/cli-input.jpg" "${CMAKE_CURRENT_BINARY_DIR}/cli-output.avif"
        --target avif --preset lossless --overwrite)
    set_tests_properties(cli_convert PROPERTIES FIXTURES_REQUIRED cli_input)
endif()
if(JXRC_BUILD_GUI)
    add_subdirectory(src/gui)
endif()
install(TARGETS jxrconvert RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} COMPONENT Runtime)
include(cmake/Packaging.cmake)
